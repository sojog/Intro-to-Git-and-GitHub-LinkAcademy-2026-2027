Pentru a vedea versiunea de git instalată
# git --version  

Pentru a vedea calea la care sunt in calculator
# pwd

Pentru a vedea elementele din folderul curent
# ls

Pentru a intra intr-un folder (accesare relativa - la pozitia actuala)
# cd NUME_FOLDER

Pentru a face un pas in spate din folder
# cd ..

Pentru a printa ceva in terminal
# echo "text de printat"

Pentru a crea un fisier cu un text in el (daca fisierul exista il suprascrie)
# echo "text de printat" > nume_fisier.terminatie

Pentru a crea un fisier cu un text in el (daca fisierul exista scrie la final)
# echo "text de printat" >> nume_fisier.terminatie

Pentru a sterge un fisier
# rm NUME_FISIER

Pentru a schimba numele fisierului
# mv NUME_ACTUAL NOUL_NUME

Pentru a schimba locatia fisierului
# mv locatia/pana/la/NUME_ACTUAL noua_locatie/pana/la/NOUL_NUME

Pentru a crea un folder
# mkdir NUME_FOLDER

Pentru a-l sterge
# rm -r NUME_FOLDER
